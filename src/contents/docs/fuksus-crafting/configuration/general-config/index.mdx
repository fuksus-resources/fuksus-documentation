---

<Note type="warning" title="Warning">
  We strongly recommend adding the image path as your inventory image path. 
  Example with ox_inventory, the path would be `nui://ox_inventory/web/images`
</Note>

```lua
Config = {}

Config.Locale = 'en'
Config.DrawDistance = 30.0  -- For object
Config.MarkerDistance = 5.0 -- For marker

Config.ImageLocation = 'nui://ox_inventory/web/images/'
Config.CraftingTickSpeed = 500 -- How often does the table check the crafting time (in ms). This does not influence crafting time

Config.Exp = {
    Exponent = 1.3,     -- How much xp next level requires, levels xp is calculated by this formula basexp * (level ^ exponent)
    Basexp = 1250,      -- How much xp is need for the firts level
    AddXpForCraft = 100 -- This is default value to add xp after crafting you can change each recipes value in the config
}

-- This is the categories for benches if no items exist in the category it will not show
Config.Categories = {
    ['weapons'] = 'Weapons',
    ['food'] = 'Food',
    ['medicine'] = 'Medicine',
    ['tools'] = 'Tools'
}

Config.SaveCraftingProgress = true -- Enable crafting progress saving
Config.SaveTime = 15               -- How often save players xp, level and crafting progress in minutes

Config.Target = 'ox_target'

Config.Debug = false

-- Global marker
Config.Marker = {
    type = 0,
    scale = {
        x = 0.5,
        y = 0.5,
        z = 0.5
    },
    color = { r = 99, g = 102, b = 241, a = 100 },
    offset = {
        x = 0.0,
        y = 0.0,
        z = 0.5
    },
    face = true,
    bobUpDown = true,
    spin = false
}
```
